   ```dart
   // apps/frontend/lib/features/customer/presentation/controllers/dashboard_controller.dart
   class CustomerDashboardController extends GetxController {
     final ApiService _apiService = ApiService.instance;
     final AuthService _authService = AuthService.instance;
     
     final RxList<Order> _recentOrders = <Order>[].obs;
     final RxList<Restaurant> _favoriteRestaurants = <Restaurant>[].obs;
     final RxList<MenuItem> _favoriteItems = <MenuItem>[].obs;
     final Rx<CustomerStats> _customerStats = CustomerStats.empty().obs;
     final RxBool _isLoading = false.obs;
     final RxBool _isRefreshing = false.obs;
     
     List<Order> get recentOrders => _recentOrders;
     List<Restaurant> get favoriteRestaurants => _favoriteRestaurants;
     List<MenuItem> get favoriteItems => _favoriteItems;
     CustomerStats get customerStats => _customerStats.value;
     bool get isLoading => _isLoading.value;
     bool get isRefreshing => _isRefreshing.value;
     
     @override
     void onInit() {
       super.onInit();
       _loadDashboardData();
       _setupRealTimeUpdates();
     }
     
     Future<void> _loadDashboardData() async {
       try {
         _isLoading.value = true;
         
         await Future.wait([
           _loadRecentOrders(),
           _loadFavoriteRestaurants(),
           _loadFavoriteItems(),
           _loadCustomerStats(),
         ]);
         
       } catch (e) {
         _handleError('Failed to load dashboard data', e);
       } finally {
         _isLoading.value = false;
       }
     }
     
     Future<void> refreshDashboard() async {
       try {
         _isRefreshing.value = true;
         await _loadDashboardData();
       } finally {
         _isRefreshing.value = false;
       }
     }
     
     Future<void> _loadRecentOrders() async {
       final response = await _apiService.get(
         '/customer/orders',
         queryParameters: {
           'limit': 10,
           'include_details': true,
         },
       );
       
       if (response.isSuccess) {
         _recentOrders.value = (response.data['orders'] as List)
             .map((json) => Order.fromJson(json))
             .toList();
       }
     }
     
     Future<void> _loadFavoriteRestaurants() async {
       final response = await _apiService.get('/customer/favorites/restaurants');
       
       if (response.isSuccess) {
         _favoriteRestaurants.value = (response.data['restaurants'] as List)
             .map((json) => Restaurant.fromJson(json))
             .toList();
       }
     }
     
     Future<void> _loadFavoriteItems() async {
       final response = await _apiService.get('/customer/favorites/items');
       
       if (response.isSuccess) {
         _favoriteItems.value = (response.data['items'] as List)
             .map((json) => MenuItem.fromJson(json))
             .toList();
       }
     }
     
     Future<void> _loadCustomerStats() async {
       final response = await _apiService.get('/customer/stats');
       
       if (response.isSuccess) {
         _customerStats.value = CustomerStats.fromJson(response.data);
       }
     }
     
     void _setupRealTimeUpdates() {
       // Listen for order status updates
       NotificationService.instance.onOrderUpdate.listen((orderUpdate) {
         _updateOrderInList(orderUpdate);
       });
       
       // Listen for loyalty point updates
       NotificationService.instance.onLoyaltyUpdate.listen((loyaltyUpdate) {
         _updateLoyaltyPoints(loyaltyUpdate);
       });
     }
     
     void _updateOrderInList(OrderUpdate update) {
       final index = _recentOrders.indexWhere((order) => order.id == update.orderId);
       if (index != -1) {
         _recentOrders[index] = _recentOrders[index].copyWith(
           status: update.status,
           updatedAt: update.updatedAt,
         );
       }
     }
     
     void _updateLoyaltyPoints(LoyaltyUpdate update) {
       _customerStats.value = _customerStats.value.copyWith(
         loyaltyPoints: update.newBalance,
       );
     }
     
     Future<void> toggleRestaurantFavorite(String restaurantId) async {
       try {
         final isFavorite = _favoriteRestaurants.any((r) => r.id == restaurantId);
         
         final response = await _apiService.post(
           '/customer/favorites/restaurants/${isFavorite ? 'remove' : 'add'}',
           data: {'restaurant_id': restaurantId},
         );
         
         if (response.isSuccess) {
           if (isFavorite) {
             _favoriteRestaurants.removeWhere((r) => r.id == restaurantId);
           } else {
             await _loadFavoriteRestaurants(); // Reload to get full data
           }
           
           SnackbarService.showSuccess(
             isFavorite ? 'Removed from favorites' : 'Added to favorites',
           );
         }
       } catch (e) {
         _handleError('Failed to update favorites', e);
       }
     }
     
     Future<void> toggleItemFavorite(String itemId) async {
       try {
         final isFavorite = _favoriteItems.any((item) => item.id == itemId);
         
         final response = await _apiService.post(
           '/customer/favorites/items/${isFavorite ? 'remove' : 'add'}',
           data: {'item_id': itemId},
         );
         
         if (response.isSuccess) {
           if (isFavorite) {
             _favoriteItems.removeWhere((item) => item.id == itemId);
           } else {
             await _loadFavoriteItems(); // Reload to get full data
           }
           
           SnackbarService.showSuccess(
             isFavorite ? 'Removed from favorites' : 'Added to favorites',
           );
         }
       } catch (e) {
         _handleError('Failed to update favorites', e);
       }
     }
     
     void navigateToOrderDetails(String orderId) {
       Get.toNamed('/orders/$orderId');
     }
     
     void navigateToRestaurant(String restaurantId) {
       Get.toNamed('/restaurants/$restaurantId');
     }
     
     void navigateToOrderHistory() {
       Get.toNamed('/orders');
     }
     
     void navigateToProfile() {
       Get.toNamed('/profile');
     }
     
     void navigateToLoyalty() {
       Get.toNamed('/loyalty');
     }
     
     void _handleError(String message, dynamic error) {
       logger.error('Dashboard Error: $message', error);
       SnackbarService.showError(message);
     }
   }
   ```

2. **Customer Stats Model**:
   ```dart
   // apps/frontend/lib/features/customer/models/customer_stats.dart
   class CustomerStats {
     final int totalOrders;
     final double totalSpent;
     final int loyaltyPoints;
     final int favoriteRestaurants;
     final int favoriteItems;
     final double averageOrderValue;
     final String memberSince;
     final CustomerTier tier;
     final List<Achievement> achievements;
     
     const CustomerStats({
       required this.totalOrders,
       required this.totalSpent,
       required this.loyaltyPoints,
       required this.favoriteRestaurants,
       required this.favoriteItems,
       required this.averageOrderValue,
     required this.memberSince,
       required this.tier,
       required this.achievements,
     });
     
     factory CustomerStats.empty() {
       return CustomerStats(
         totalOrders: 0,
         totalSpent: 0.0,
         loyaltyPoints: 0,
         favoriteRestaurants: 0,
         favoriteItems: 0,
         averageOrderValue: 0.0,
         memberSince: '',
         tier: CustomerTier.bronze,
         achievements: [],
       );
     }
     
     factory CustomerStats.fromJson(Map<String, dynamic> json) {
       return CustomerStats(
         totalOrders: json['total_orders'] ?? 0,
         totalSpent: (json['total_spent'] ?? 0).toDouble(),
         loyaltyPoints: json['loyalty_points'] ?? 0,
         favoriteRestaurants: json['favorite_restaurants'] ?? 0,
         favoriteItems: json['favorite_items'] ?? 0,
         averageOrderValue: (json['average_order_value'] ?? 0).toDouble(),
         memberSince: json['member_since'] ?? '',
         tier: CustomerTier.fromString(json['tier'] ?? 'bronze'),
         achievements: (json['achievements'] as List? ?? [])
             .map((json) => Achievement.fromJson(json))
             .toList(),
       );
     }
     
     CustomerStats copyWith({
       int? totalOrders,
       double? totalSpent,
       int? loyaltyPoints,
       int? favoriteRestaurants,
       int? favoriteItems,
       double? averageOrderValue,
       String? memberSince,
       CustomerTier? tier,
       List<Achievement>? achievements,
     }) {
       return CustomerStats(
         totalOrders: totalOrders ?? this.totalOrders,
         totalSpent: totalSpent ?? this.totalSpent,
         loyaltyPoints: loyaltyPoints ?? this.loyaltyPoints,
         favoriteRestaurants: favoriteRestaurants ?? this.favoriteRestaurants,
         favoriteItems: favoriteItems ?? this.favoriteItems,
         averageOrderValue: averageOrderValue ?? this.averageOrderValue,
         memberSince: memberSince ?? this.memberSince,
         tier: tier ?? this.tier,
         achievements: achievements ?? this.achievements,
       );
     }
   }
   
   enum CustomerTier {
     bronze,
     silver,
     gold,
     platinum;
     
     static CustomerTier fromString(String value) {
       return CustomerTier.values.firstWhere(
         (tier) => tier.name.toLowerCase() == value.toLowerCase(),
         orElse: () => CustomerTier.bronze,
       );
     }
     
     Color get color {
       switch (this) {
         case CustomerTier.bronze:
           return Colors.brown;
         case CustomerTier.silver:
           return Colors.grey;
         case CustomerTier.gold:
           return Colors.amber;
         case CustomerTier.platinum:
           return Colors.cyan;
       }
     }
     
     String get displayName {
       return name.substring(0, 1).toUpperCase() + name.substring(1);
     }
   }
   ```
