   ```python
   # apps/backend/app/features/notifications/infrastructure/notification_service.py
   from abc import ABC, abstractmethod
   from enum import Enum
   from typing import Dict, List, Optional, Any
   import asyncio
   from datetime import datetime, timedelta
   import json
   
   class NotificationType(Enum):
       PUSH = "push"
       SMS = "sms"
       EMAIL = "email"
       IN_APP = "in_app"
       WEBHOOK = "webhook"
   
   class NotificationPriority(Enum):
       LOW = "low"
       MEDIUM = "medium"
       HIGH = "high"
       URGENT = "urgent"
   
   class NotificationChannel(ABC):
       @abstractmethod
       async def send(self, notification: 'Notification') -> bool:
           pass
       
       @abstractmethod
       async def validate_recipient(self, recipient: str) -> bool:
           pass
   
   class PushNotificationChannel(NotificationChannel):
       def __init__(self, fcm_server_key: str):
           self.fcm_server_key = fcm_server_key
           
       async def send(self, notification: 'Notification') -> bool:
           """Send FCM push notification"""
           import aiohttp
           
           fcm_url = "https://fcm.googleapis.com/fcm/send"
           headers = {
               "Authorization": f"key={self.fcm_server_key}",
               "Content-Type": "application/json"
           }
           
           payload = {
               "to": notification.recipient,
               "notification": {
                   "title": notification.title,
                   "body": notification.content,
                   "icon": notification.metadata.get('icon', 'default'),
                   "sound": notification.metadata.get('sound', 'default'),
                   "click_action": notification.metadata.get('click_action')
               },
               "data": {
                   "notification_id": notification.id,
                   "type": notification.notification_type,
                   "priority": notification.priority.value,
                   **notification.metadata
               },
               "priority": "high" if notification.priority in [NotificationPriority.HIGH, NotificationPriority.URGENT] else "normal"
           }
           
           try:
               async with aiohttp.ClientSession() as session:
                   async with session.post(fcm_url, json=payload, headers=headers) as response:
                       if response.status == 200:
                           result = await response.json()
                           return result.get('success', 0) > 0
                       return False
           except Exception as e:
               logger.error(f"FCM notification failed: {e}")
               return False
   
   class Notification:
       def __init__(
           self,
           id: str,
           recipient: str,
           channel: NotificationType,
           title: str,
           content: str,
           notification_type: str,
           priority: NotificationPriority = NotificationPriority.MEDIUM,
           metadata: Optional[Dict] = None,
           scheduled_at: Optional[datetime] = None
       ):
           self.id = id
           self.recipient = recipient
           self.channel = channel
           self.title = title
           self.content = content
           self.notification_type = notification_type
           self.priority = priority
           self.metadata = metadata or {}
           self.scheduled_at = scheduled_at or datetime.utcnow()
           self.created_at = datetime.utcnow()
           self.retry_count = 0
           self.max_retries = 3
   
   class NotificationService:
       def __init__(self, redis_client, database):
           self.redis = redis_client
           self.db = database
           self.channels = {}
           self.templates = {}
           
       def register_channel(self, channel_type: NotificationType, channel: NotificationChannel):
           """Register a notification channel"""
           self.channels[channel_type] = channel
           
       async def send_notification(
           self,
           recipient: str,
           channel: NotificationType,
           template_id: str,
           variables: Dict = None,
           priority: NotificationPriority = NotificationPriority.MEDIUM,
           scheduled_at: Optional[datetime] = None,
           metadata: Dict = None
       ) -> str:
           """Send a notification using a template"""
           
           # Get template
           template = self.templates.get(template_id)
           if not template:
               raise ValueError(f"Template {template_id} not found")
           
           # Render template with variables
           title = self._render_template(template['title'], variables or {})
           content = self._render_template(template['content'], variables or {})
           
           # Create notification
           notification = Notification(
               id=self._generate_notification_id(),
               recipient=recipient,
               channel=channel,
               title=title,
               content=content,
               notification_type=template_id,
               priority=priority,
               metadata=metadata,
               scheduled_at=scheduled_at
           )
           
           # Queue for immediate or scheduled delivery
           if scheduled_at and scheduled_at > datetime.utcnow():
               await self._schedule_notification(notification)
           else:
               await self._queue_notification(notification)
           
           return notification.id
   ```

**Phase 2: Real-time Notification Processing**

1. **Notification Queue Processor**:
   ```python
   # apps/backend/app/features/notifications/application/notification_processor.py
   import asyncio
   import json
   from typing import Dict, List
   from datetime import datetime, timedelta
   
   class NotificationProcessor:
       def __init__(self, notification_service):
           self.notification_service = notification_service
           self.processing = False
           
       async def start_processing(self):
           """Start processing notification queue"""
           self.processing = True
           
           # Process different priority queues
           priority_queues = [
               NotificationPriority.URGENT,
               NotificationPriority.HIGH,
               NotificationPriority.MEDIUM,
               NotificationPriority.LOW
           ]
           
           while self.processing:
               try:
                   for priority in priority_queues:
                       queue_key = f"notifications:queue:{priority.value}"
                       
                       # Process batch of notifications
                       batch = await self.notification_service.redis.lrange(queue_key, 0, 9)
                       if batch:
                           await self.notification_service.redis.ltrim(queue_key, len(batch), -1)
                           
                           # Process notifications in parallel
                           tasks = []
                           for notification_data in batch:
                               notification_dict = json.loads(notification_data)
                               task = self._process_single_notification(notification_dict)
                               tasks.append(task)
                           
                           await asyncio.gather(*tasks, return_exceptions=True)
                   
                   # Check for scheduled notifications
                   await self._process_scheduled_notifications()
                   
                   # Wait before next iteration
                   await asyncio.sleep(1)
                   
               except Exception as e:
                   logger.error(f"Error processing notification queue: {e}")
                   await asyncio.sleep(5)
                   
       async def _process_single_notification(self, notification_dict: Dict):
           """Process a single notification"""
           try:
               channel_type = NotificationType(notification_dict['channel'])
               channel = self.notification_service.channels.get(channel_type)
               
               if not channel:
                   logger.error(f"Channel {channel_type} not registered")
                   return
               
               # Create notification object
               notification = Notification(
                   id=notification_dict['id'],
                   recipient=notification_dict['recipient'],
                   channel=channel_type,
                   title=notification_dict['title'],
                   content=notification_dict['content'],
                   notification_type=notification_dict['notification_type'],
                   priority=NotificationPriority(notification_dict['priority']),
                   metadata=json.loads(notification_dict['metadata'])
               )
               
               # Send notification
               success = await channel.send(notification)
               
               if success:
                   await self._update_notification_status(notification, "sent")
               else:
                   await self._handle_failed_notification(notification)
                   
           except Exception as e:
               logger.error(f"Error processing notification: {e}")
   ```

1. **Flutter Notification Handler**:
   ```dart
   // apps/frontend/lib/core/services/notification_service.dart
   class NotificationService extends GetxService {
     static NotificationService get instance => Get.find();
     
     late FlutterLocalNotificationsPlugin _localNotifications;
     late FirebaseMessaging _firebaseMessaging;
     
     final RxList<AppNotification> _notifications = <AppNotification>[].obs;
     final RxInt _unreadCount = 0.obs;
     
     List<AppNotification> get notifications => _notifications;
     int get unreadCount => _unreadCount.value;
     
     @override
     Future<void> onInit() async {
       super.onInit();
       await _initializeLocalNotifications();
       await _initializeFirebaseMessaging();
       await _setupNotificationHandlers();
     }
     
     Future<void> _initializeFirebaseMessaging() async {
       _firebaseMessaging = FirebaseMessaging.instance;
       
       // Request permission
       NotificationSettings settings = await _firebaseMessaging.requestPermission(
         alert: true,
         badge: true,
         sound: true,
         provisional: false,
       );
       
       if (settings.authorizationStatus == AuthorizationStatus.authorized) {
         // Get FCM token
         String? token = await _firebaseMessaging.getToken();
         if (token != null) {
           await _updateFCMToken(token);
         }
         
         // Listen for token refresh
         _firebaseMessaging.onTokenRefresh.listen(_updateFCMToken);
       }
     }
     
     Future<void> _setupNotificationHandlers() async {
       // Handle foreground messages
       FirebaseMessaging.onMessage.listen(_handleForegroundMessage);
       
       // Handle notification when app is opened from background
       FirebaseMessaging.onMessageOpenedApp.listen(_handleNotificationOpened);
       
       // Handle notification when app is opened from terminated state
       RemoteMessage? initialMessage = await _firebaseMessaging.getInitialMessage();
       if (initialMessage != null) {
         _handleNotificationOpened(initialMessage);
       }
     }
     
     Future<void> _handleForegroundMessage(RemoteMessage message) async {
       // Show local notification when app is in foreground
       await _showLocalNotification(
         title: message.notification?.title ?? 'Notification',
         body: message.notification?.body ?? '',
         payload: jsonEncode(message.data),
       );
       
       // Add to in-app notification list
       final notification = AppNotification.fromRemoteMessage(message);
       _notifications.insert(0, notification);
       _unreadCount.value++;
     }
     
     void _handleNotificationOpened(RemoteMessage message) {
       _navigateBasedOnNotification(message.data);
     }
     
     void _navigateBasedOnNotification(Map<String, dynamic> data) {
       final clickAction = data['click_action'] as String?;
       
       switch (clickAction) {
         case 'ORDER_DETAILS':
           Get.toNamed('/orders/${data['order_id']}');
           break;
         case 'ORDER_MANAGEMENT':
           Get.toNamed('/restaurant/orders');
           break;
         case 'ANALYTICS_DASHBOARD':
           Get.toNamed('/restaurant/analytics');
           break;
         default:
           Get.toNamed('/notifications');
       }
     }
     
     Future<void> markAsRead(String notificationId) async {
       final index = _notifications.indexWhere((n) => n.id == notificationId);
       if (index != -1 && !_notifications[index].isRead) {
         _notifications[index] = _notifications[index].copyWith(isRead: true);
         _unreadCount.value = _notifications.where((n) => !n.isRead).length;
         
         // Update on server
         await ApiService.instance.markNotificationAsRead(notificationId);
       }
     }
   }
   ```

   ```python
   from fastapi import APIRouter, Depends, Query
   from typing import List, Optional
   
   router = APIRouter(prefix="/api/notifications", tags=["notifications"])
   
   @router.get("/restaurant/{restaurant_id}")
   async def get_restaurant_notifications(
       restaurant_id: str,
       status: Optional[str] = Query(None),
       limit: int = Query(50, le=100),
       current_user: User = Depends(get_current_restaurant_user)
   ):
       """Get notifications for restaurant staff"""
       
       if not await verify_restaurant_access(current_user.id, restaurant_id):
           raise HTTPException(status_code=403, detail="Access denied")
       
       notification_service = NotificationService()
       notifications = await notification_service.get_restaurant_notifications(
           restaurant_id=restaurant_id,
           status=status,
           limit=limit
       )
       
       return {
           "notifications": notifications,
           "total_count": len(notifications),
           "unread_count": len([n for n in notifications if not n.get('read', False)])
       }
   
   @router.post("/send-bulk")
   async def send_bulk_notifications(

       template_id: str,
   ```
