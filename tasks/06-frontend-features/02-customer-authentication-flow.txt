**Objective**: Create a comprehensive customer authentication system for the ZERGO QR platform with secure JWT-based authentication, biometric login options, social authentication, and seamless user experience across mobile and web platforms.

**Phase 1: Authentication Architecture**

1. **Flutter Authentication Service**:
   ```dart
   // apps/frontend/lib/core/services/auth_service.dart
   class AuthService extends GetxService {
     static AuthService get instance => Get.find();
     
     final Rx<User?> _currentUser = Rx<User?>(null);
     final RxBool _isAuthenticated = false.obs;
     final RxBool _isLoading = false.obs;
     final RxString _authToken = ''.obs;
     
     User? get currentUser => _currentUser.value;
     bool get isAuthenticated => _isAuthenticated.value;
     bool get isLoading => _isLoading.value;
     String get authToken => _authToken.value;
     
     @override
     Future<void> onInit() async {
       super.onInit();
       await _initializeAuth();
     }
     
     Future<void> _initializeAuth() async {
       // Check for stored auth token
       final storedToken = await StorageService.instance.getAuthToken();
       if (storedToken != null) {
         _authToken.value = storedToken;
         await _validateAndRefreshToken();
       }
     }
     
     Future<AuthResult> loginWithEmail({
       required String email,
       required String password,
       bool rememberMe = false,
     }) async {
       try {
         _isLoading.value = true;
         
         final response = await ApiService.instance.post(
           '/auth/login',
           data: {
             'email': email,
             'password': password,
             'device_info': await _getDeviceInfo(),
           },
         );
         
         if (response.isSuccess) {
           final authData = AuthData.fromJson(response.data);
           await _handleSuccessfulAuth(authData, rememberMe);
           
           return AuthResult.success(
             user: authData.user,
             message: 'Login successful',
           );
         } else {
           return AuthResult.error(
             message: response.message ?? 'Login failed',
             errorCode: response.errorCode,
           );
         }
       } catch (e) {
         return AuthResult.error(
           message: 'An unexpected error occurred',
           errorCode: 'UNKNOWN_ERROR',
         );
       } finally {
         _isLoading.value = false;
       }
     }
     
     Future<AuthResult> registerWithEmail({
       required String email,
       required String password,
       required String fullName,
       required String phoneNumber,
       String? referralCode,
     }) async {
       try {
         _isLoading.value = true;
         
         final response = await ApiService.instance.post(
           '/auth/register',
           data: {
             'email': email,
             'password': password,
             'full_name': fullName,
             'phone_number': phoneNumber,
             'referral_code': referralCode,
             'device_info': await _getDeviceInfo(),
           },
         );
         
         if (response.isSuccess) {
           // Send OTP for verification
           await _sendVerificationOTP(email);
           
           return AuthResult.success(
             message: 'Registration successful. Please verify your email.',
             requiresVerification: true,
           );
         } else {
           return AuthResult.error(
             message: response.message ?? 'Registration failed',
             errorCode: response.errorCode,
           );
         }
       } catch (e) {
         return AuthResult.error(
           message: 'Registration failed',
           errorCode: 'REGISTRATION_ERROR',
         );
       } finally {
         _isLoading.value = false;
       }
     }
     
     Future<AuthResult> loginWithBiometrics() async {
       try {
         _isLoading.value = true;
         
         // Check if biometrics are available
         final isAvailable = await BiometricService.instance.isAvailable();
         if (!isAvailable) {
           return AuthResult.error(
             message: 'Biometric authentication not available',
             errorCode: 'BIOMETRICS_UNAVAILABLE',
           );
         }
         
         // Get stored biometric credentials
         final biometricData = await BiometricService.instance.getStoredCredentials();
         if (biometricData == null) {
           return AuthResult.error(
             message: 'No biometric credentials found',
             errorCode: 'NO_BIOMETRIC_CREDENTIALS',
           );
         }
         
         // Authenticate with biometrics
         final biometricResult = await BiometricService.instance.authenticate(
           reason: 'Authenticate to access your account',
         );
         
         if (biometricResult.isSuccess) {
           // Login with stored credentials
           final response = await ApiService.instance.post(
             '/auth/biometric-login',
             data: {
               'biometric_token': biometricData.token,
               'device_id': await DeviceService.instance.getDeviceId(),
             },
           );
           
           if (response.isSuccess) {
             final authData = AuthData.fromJson(response.data);
             await _handleSuccessfulAuth(authData, true);
             
             return AuthResult.success(
               user: authData.user,
               message: 'Biometric login successful',
             );
           }
         }
         
         return AuthResult.error(
           message: 'Biometric authentication failed',
           errorCode: 'BIOMETRIC_AUTH_FAILED',
         );
       } catch (e) {
         return AuthResult.error(
           message: 'Biometric login failed',
           errorCode: 'BIOMETRIC_ERROR',
         );
       } finally {
         _isLoading.value = false;
       }
     }
     
     Future<AuthResult> loginWithGoogle() async {
       try {
         _isLoading.value = true;
         
         final googleUser = await GoogleSignIn().signIn();
         if (googleUser == null) {
           return AuthResult.error(
             message: 'Google sign-in cancelled',
             errorCode: 'GOOGLE_SIGNIN_CANCELLED',
           );
         }
         
         final googleAuth = await googleUser.authentication;
         
         final response = await ApiService.instance.post(
           '/auth/google-login',
           data: {
             'google_token': googleAuth.accessToken,
             'id_token': googleAuth.idToken,
             'device_info': await _getDeviceInfo(),
           },
         );
         
         if (response.isSuccess) {
           final authData = AuthData.fromJson(response.data);
           await _handleSuccessfulAuth(authData, true);
           
           return AuthResult.success(
             user: authData.user,
             message: 'Google login successful',
           );
         } else {
           return AuthResult.error(
             message: response.message ?? 'Google login failed',
             errorCode: response.errorCode,
           );
         }
       } catch (e) {
         return AuthResult.error(
           message: 'Google login failed',
           errorCode: 'GOOGLE_LOGIN_ERROR',
         );
       } finally {
         _isLoading.value = false;
       }
     }
     
     Future<AuthResult> verifyOTP({
       required String email,
       required String otp,
     }) async {
       try {
         _isLoading.value = true;
         
         final response = await ApiService.instance.post(
           '/auth/verify-otp',
           data: {
             'email': email,
             'otp': otp,
           },
         );
         
         if (response.isSuccess) {
           final authData = AuthData.fromJson(response.data);
           await _handleSuccessfulAuth(authData, true);
           
           return AuthResult.success(
             user: authData.user,
             message: 'Email verified successfully',
           );
         } else {
           return AuthResult.error(
             message: response.message ?? 'OTP verification failed',
             errorCode: response.errorCode,
           );
         }
       } catch (e) {
         return AuthResult.error(
           message: 'OTP verification failed',
           errorCode: 'OTP_VERIFICATION_ERROR',
         );
       } finally {
         _isLoading.value = false;
       }
     }
     
     Future<void> _handleSuccessfulAuth(AuthData authData, bool rememberMe) async {
       _authToken.value = authData.accessToken;
       _currentUser.value = authData.user;
       _isAuthenticated.value = true;
       
       // Store tokens securely
       if (rememberMe) {
         await StorageService.instance.setAuthToken(authData.accessToken);
         await StorageService.instance.setRefreshToken(authData.refreshToken);
       }
       
       // Update FCM token for notifications
       await NotificationService.instance.updateFCMToken();
       
       // Initialize user-specific services
       await _initializeUserServices();
     }
     
     Future<void> logout() async {
       try {
         // Notify server about logout
         await ApiService.instance.post('/auth/logout');
       } catch (e) {
         // Continue with local logout even if server request fails
       }
       
       // Clear local auth state
       _authToken.value = '';
       _currentUser.value = null;
       _isAuthenticated.value = false;
       
       // Clear stored tokens
       await StorageService.instance.clearAuthData();
       
       // Clear user-specific data
       await _clearUserServices();
       
       // Navigate to login
       Get.offAllNamed('/login');
     }
     
     Future<bool> _validateAndRefreshToken() async {
       try {
         final response = await ApiService.instance.get('/auth/validate-token');
         
         if (response.isSuccess) {
           final userData = User.fromJson(response.data['user']);
           _currentUser.value = userData;
           _isAuthenticated.value = true;
           await _initializeUserServices();
           return true;
         } else {
           // Try to refresh token
           return await _refreshAuthToken();
         }
       } catch (e) {
         return false;
       }
     }
     
     Future<bool> _refreshAuthToken() async {
       try {
         final refreshToken = await StorageService.instance.getRefreshToken();
         if (refreshToken == null) return false;
         
         final response = await ApiService.instance.post(
           '/auth/refresh-token',
           data: {'refresh_token': refreshToken},
         );
         
         if (response.isSuccess) {
           final authData = AuthData.fromJson(response.data);
           await _handleSuccessfulAuth(authData, true);
           return true;
         } else {
           await logout();
           return false;
         }
       } catch (e) {
         await logout();
         return false;
       }
     }
   }
   ```

   ```dart
   // apps/frontend/lib/features/auth/models/auth_models.dart
   class User {
     final String id;
     final String email;
     final String fullName;
     final String? phoneNumber;
     final String? profileImageUrl;
     final DateTime createdAt;
     final DateTime? lastLoginAt;
     final bool isEmailVerified;
     final bool isPhoneVerified;
     final UserPreferences preferences;
     
     const User({
       required this.id,
       required this.email,
       required this.fullName,
       this.phoneNumber,
       this.profileImageUrl,
       required this.createdAt,
       this.lastLoginAt,
       required this.isEmailVerified,
       required this.isPhoneVerified,
       required this.preferences,
     });
     
     factory User.fromJson(Map<String, dynamic> json) {
       return User(
         id: json['id'],
         email: json['email'],
         fullName: json['full_name'],
         phoneNumber: json['phone_number'],
         profileImageUrl: json['profile_image_url'],
         createdAt: DateTime.parse(json['created_at']),
         lastLoginAt: json['last_login_at'] != null 
             ? DateTime.parse(json['last_login_at']) 
             : null,
         isEmailVerified: json['is_email_verified'] ?? false,
         isPhoneVerified: json['is_phone_verified'] ?? false,
         preferences: UserPreferences.fromJson(json['preferences'] ?? {}),
       );
     }
   }
   
   class AuthData {
     final String accessToken;
     final String refreshToken;
     final User user;
     final DateTime expiresAt;
     
     const AuthData({
       required this.accessToken,
       required this.refreshToken,
       required this.user,
       required this.expiresAt,
     });
     
     factory AuthData.fromJson(Map<String, dynamic> json) {
       return AuthData(
         accessToken: json['access_token'],
         refreshToken: json['refresh_token'],
         user: User.fromJson(json['user']),
         expiresAt: DateTime.parse(json['expires_at']),
       );
     }
   }
   
   class AuthResult {
     final bool isSuccess;
     final User? user;
     final String? message;
     final String? errorCode;
     final bool requiresVerification;
     
     const AuthResult._({
       required this.isSuccess,
       this.user,
       this.message,
       this.errorCode,
       this.requiresVerification = false,
     });
     
     factory AuthResult.success({
       User? user,
       String? message,
       bool requiresVerification = false,
     }) {
       return AuthResult._(
         isSuccess: true,
         user: user,
         message: message,
         requiresVerification: requiresVerification,
       );
     }
     
     factory AuthResult.error({
       required String message,
       String? errorCode,
     }) {
       return AuthResult._(
         isSuccess: false,
         message: message,
         errorCode: errorCode,
       );
     }
   }
   ```
