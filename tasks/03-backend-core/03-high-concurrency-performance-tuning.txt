   ```python
   # app/core/async_processor.py
   import asyncio
   import aioredis
   import aioboto3
   from concurrent.futures import ThreadPoolExecutor
   from typing import Dict, List, Any, Callable, Optional
   import logging
   import time
   from dataclasses import dataclass
   from enum import Enum
   
   logger = logging.getLogger(__name__)
   
   class TaskPriority(Enum):
       LOW = 1
       MEDIUM = 2
       HIGH = 3
       URGENT = 4
   
   @dataclass
   class AsyncTask:
       id: str
       func: Callable
       args: tuple
       kwargs: dict
       priority: TaskPriority
       max_retries: int = 3
       retry_count: int = 0
       created_at: float = time.time()
   
   class HighConcurrencyProcessor:
       def __init__(self, max_workers: int = 50, redis_url: str = None):
           self.max_workers = max_workers
           self.executor = ThreadPoolExecutor(max_workers=max_workers)
           self.task_queues = {
               TaskPriority.URGENT: asyncio.PriorityQueue(),
               TaskPriority.HIGH: asyncio.PriorityQueue(),
               TaskPriority.MEDIUM: asyncio.PriorityQueue(),
               TaskPriority.LOW: asyncio.PriorityQueue()
           }
           self.active_tasks = {}
           self.completed_tasks = {}
           self.failed_tasks = {}
           self.redis = None
           self.redis_url = redis_url
           self.worker_pools = {}
           self.semaphores = {}
           
       async def initialize(self):
           """Initialize async processor with Redis connection"""
           if self.redis_url:
               self.redis = await aioredis.from_url(self.redis_url, decode_responses=True)
           
           # Create semaphores for different operation types
           self.semaphores = {
               'database': asyncio.Semaphore(20),  # Max 20 concurrent DB operations
               'api_calls': asyncio.Semaphore(30),  # Max 30 concurrent API calls
               'file_operations': asyncio.Semaphore(10),  # Max 10 concurrent file ops
               'email_sending': asyncio.Semaphore(5),  # Max 5 concurrent emails
               'sms_sending': asyncio.Semaphore(3)  # Max 3 concurrent SMS
           }
           
           # Start worker pools
           for priority in TaskPriority:
               self.worker_pools[priority] = [
                   asyncio.create_task(self._worker(priority, i))
                   for i in range(max(1, self.max_workers // len(TaskPriority)))
               ]
               
       async def _worker(self, priority: TaskPriority, worker_id: int):
           """Worker coroutine for processing tasks"""
           queue = self.task_queues[priority]
           
           while True:
               try:
                   # Get task from queue (with timeout to allow graceful shutdown)
                   try:
                       priority_value, task = await asyncio.wait_for(queue.get(), timeout=1.0)
                   except asyncio.TimeoutError:
                       continue
                   
                   # Execute task
                   await self._execute_task(task, worker_id)
                   queue.task_done()
                   
               except asyncio.CancelledError:
                   break
               except Exception as e:
                   logger.error(f"Worker {worker_id} error: {e}")
                   
       async def _execute_task(self, task: AsyncTask, worker_id: int):
           """Execute individual task with proper resource management"""
           task_start = time.time()
           
           try:
               # Determine appropriate semaphore based on task type
               semaphore = self._get_task_semaphore(task)
               
               async with semaphore:
                   self.active_tasks[task.id] = {
                       'task': task,
                       'worker_id': worker_id,
                       'start_time': task_start
                   }
                   
                   # Execute task
                   if asyncio.iscoroutinefunction(task.func):
                       result = await task.func(*task.args, **task.kwargs)
                   else:
                       # Run CPU-bound tasks in thread pool
                       result = await asyncio.get_event_loop().run_in_executor(
                           self.executor, task.func, *task.args
                       )
                   
                   # Mark as completed
                   execution_time = time.time() - task_start
                   self.completed_tasks[task.id] = {
                       'result': result,
                       'execution_time': execution_time,
                       'completed_at': time.time()
                   }
                   
                   # Remove from active tasks
                   del self.active_tasks[task.id]
                   
                   logger.debug(f"Task {task.id} completed in {execution_time:.3f}s by worker {worker_id}")
                   
           except Exception as e:
               # Handle task failure
               await self._handle_task_failure(task, e)
               
       def _get_task_semaphore(self, task: AsyncTask) -> asyncio.Semaphore:
           """Determine appropriate semaphore based on task function name"""
           func_name = task.func.__name__
           
           if 'database' in func_name or 'db_' in func_name or 'query' in func_name:
               return self.semaphores['database']
           elif 'api_' in func_name or 'http_' in func_name or 'request' in func_name:
               return self.semaphores['api_calls']
           elif 'file_' in func_name or 'upload' in func_name or 'download' in func_name:
               return self.semaphores['file_operations']
           elif 'email' in func_name or 'mail' in func_name:
               return self.semaphores['email_sending']
           elif 'sms' in func_name or 'text' in func_name:
               return self.semaphores['sms_sending']
           else:
               return self.semaphores['api_calls']  # Default
               
       async def _handle_task_failure(self, task: AsyncTask, error: Exception):
           """Handle task failure with retry logic"""
           task.retry_count += 1
           
           if task.retry_count <= task.max_retries:
               # Exponential backoff
               delay = min(300, 2 ** task.retry_count)  # Max 5 minutes
               await asyncio.sleep(delay)
               
               # Re-queue task
               await self.add_task(
                   task.func, *task.args, 
                   priority=task.priority,
                   task_id=task.id,
                   max_retries=task.max_retries,
                   **task.kwargs
               )
               
               logger.warning(f"Retrying task {task.id} (attempt {task.retry_count}/{task.max_retries})")
           else:
               # Max retries exceeded
               self.failed_tasks[task.id] = {
                   'task': task,
                   'error': str(error),
                   'failed_at': time.time()
               }
               
               # Remove from active tasks
               if task.id in self.active_tasks:
                   del self.active_tasks[task.id]
               
               logger.error(f"Task {task.id} failed permanently: {error}")
               
       async def add_task(
           self, 
           func: Callable, 
           *args, 
           priority: TaskPriority = TaskPriority.MEDIUM,
           task_id: str = None,
           max_retries: int = 3,
           **kwargs
       ) -> str:
           """Add task to processing queue"""
           if task_id is None:
               task_id = f"task_{int(time.time() * 1000000)}"
           
           task = AsyncTask(
               id=task_id,
               func=func,
               args=args,
               kwargs=kwargs,
               priority=priority,
               max_retries=max_retries
           )
           
           # Add to appropriate queue (negative priority for max heap behavior)
           await self.task_queues[priority].put((-priority.value, task))
           
           logger.debug(f"Added task {task_id} with priority {priority.name}")
           return task_id
           
       async def get_task_status(self, task_id: str) -> Dict[str, Any]:
           """Get status of a specific task"""
           if task_id in self.active_tasks:
               task_info = self.active_tasks[task_id]
               return {
                   'status': 'running',
                   'worker_id': task_info['worker_id'],
                   'running_time': time.time() - task_info['start_time']
               }
           elif task_id in self.completed_tasks:
               return {
                   'status': 'completed',
                   'result': self.completed_tasks[task_id]['result'],
                   'execution_time': self.completed_tasks[task_id]['execution_time']
               }
           elif task_id in self.failed_tasks:
               return {
                   'status': 'failed',
                   'error': self.failed_tasks[task_id]['error']
               }
           else:
               return {'status': 'not_found'}
               
       async def get_system_stats(self) -> Dict[str, Any]:
           """Get comprehensive system performance statistics"""
           queue_sizes = {
               priority.name: self.task_queues[priority].qsize()
               for priority in TaskPriority
           }
           
           return {
               'active_tasks': len(self.active_tasks),
               'completed_tasks': len(self.completed_tasks),
               'failed_tasks': len(self.failed_tasks),
               'queue_sizes': queue_sizes,
               'worker_count': sum(len(workers) for workers in self.worker_pools.values()),
               'semaphore_usage': {
                   name: sem._value for name, sem in self.semaphores.items()
               },
               'memory_usage': await self._get_memory_usage(),
               'thread_pool_usage': {
                   'active_threads': self.executor._threads,
                   'max_workers': self.executor._max_workers
               }
           }
           
       async def _get_memory_usage(self) -> Dict[str, Any]:
           """Get current memory usage statistics"""
           import psutil
           import os
           
           process = psutil.Process(os.getpid())
           memory_info = process.memory_info()
           
           return {
               'rss_mb': memory_info.rss / 1024 / 1024,
               'vms_mb': memory_info.vms / 1024 / 1024,
               'cpu_percent': process.cpu_percent(),
               'num_threads': process.num_threads()
           }
           
       async def shutdown(self):
           """Graceful shutdown of async processor"""
           logger.info("Shutting down async processor...")
           
           # Cancel all workers
           for workers in self.worker_pools.values():
               for worker in workers:
                   worker.cancel()
           
           # Wait for workers to finish
           for workers in self.worker_pools.values():
               await asyncio.gather(*workers, return_exceptions=True)
           
           # Shutdown thread pool
           self.executor.shutdown(wait=True)
           
           # Close Redis connection
           if self.redis:
               await self.redis.close()
               
           logger.info("Async processor shutdown complete")
   ```

   ```python
   # app/core/db_pool_manager.py
   import asyncio
   import asyncpg
   import time
   from typing import Dict, List, Optional, Any
   import logging
   from contextlib import asynccontextmanager
   from dataclasses import dataclass
   
   logger = logging.getLogger(__name__)
   
   @dataclass
   class ConnectionStats:
       total_connections: int
       active_connections: int
       idle_connections: int
       avg_query_time: float
       total_queries: int
       failed_queries: int
       
   class OptimizedDBPoolManager:
       def __init__(self, database_url: str):
           self.database_url = database_url
           self.pool = None
           self.connection_stats = {}
           self.query_metrics = []
           self.max_query_time = 0
           self.total_queries = 0
           self.failed_queries = 0
           
       async def initialize_pool(
           self,
           min_size: int = 10,
           max_size: int = 50,
           max_queries: int = 50000,
           max_inactive_connection_lifetime: float = 300.0,
           command_timeout: float = 30.0
       ):
           """Initialize optimized connection pool"""
           
           async def connection_init(conn):
               """Initialize each connection with optimizations"""
               # Set connection-level optimizations
               await conn.execute('SET statement_timeout = $1', command_timeout * 1000)
               await conn.execute('SET idle_in_transaction_session_timeout = $1', 60000)
               await conn.execute('SET tcp_keepalives_idle = $1', 600)
               await conn.execute('SET tcp_keepalives_interval = $1', 30)
               await conn.execute('SET tcp_keepalives_count = $1', 3)
               
               # Prepare frequently used statements
               await conn.prepare('SELECT 1')  # Health check
               
           try:
               self.pool = await asyncpg.create_pool(
                   self.database_url,
                   min_size=min_size,
                   max_size=max_size,
                   max_queries=max_queries,
                   max_inactive_connection_lifetime=max_inactive_connection_lifetime,
                   command_timeout=command_timeout,
                   init=connection_init,
                   setup=self._setup_connection
               )
               
               logger.info(f"Database pool initialized: {min_size}-{max_size} connections")
               
               # Start monitoring task
               asyncio.create_task(self._monitor_pool())
               
           except Exception as e:
               logger.error(f"Failed to initialize database pool: {e}")
               raise
               
       async def _setup_connection(self, conn):
           """Additional connection setup for performance"""
           # Set application name for monitoring
           await conn.execute("SET application_name = 'zergo_qr_api'")
           
           # Enable parallel queries where possible
           await conn.execute("SET max_parallel_workers_per_gather = 2")
           
       @asynccontextmanager
       async def get_connection(self):
           """Get connection from pool with metrics"""
           start_time = time.time()
           connection = None
           
           try:
               connection = await self.pool.acquire()
               acquire_time = time.time() - start_time
               
               if acquire_time > 0.1:  # Log slow acquisitions
                   logger.warning(f"Slow connection acquisition: {acquire_time:.3f}s")
               
               yield connection
               
           except Exception as e:
               self.failed_queries += 1
               logger.error(f"Database connection error: {e}")
               raise
           finally:
               if connection:
                   await self.pool.release(connection)
                   
       async def execute_query_with_metrics(
           self, 
           query: str, 
           *args, 
           fetch_all: bool = False,
           fetch_one: bool = False
       ) -> Any:
           """Execute query with performance metrics"""
           start_time = time.time()
           
           try:
               async with self.get_connection() as conn:
                   if fetch_all:
                       result = await conn.fetch(query, *args)
                   elif fetch_one:
                       result = await conn.fetchrow(query, *args)
                   else:
                       result = await conn.execute(query, *args)
                   
                   # Record metrics
                   execution_time = time.time() - start_time
                   self.total_queries += 1
                   self.query_metrics.append(execution_time)
                   
                   # Keep only last 1000 query times for rolling average
                   if len(self.query_metrics) > 1000:
                       self.query_metrics = self.query_metrics[-1000:]
                   
                   if execution_time > self.max_query_time:
                       self.max_query_time = execution_time
                   
                   # Log slow queries
                   if execution_time > 1.0:
                       logger.warning(f"Slow query ({execution_time:.3f}s): {query[:100]}...")
                   
                   return result
                   
           except Exception as e:
               self.failed_queries += 1
               execution_time = time.time() - start_time
               logger.error(f"Query failed after {execution_time:.3f}s: {e}")
               raise
               
       async def execute_transaction(self, queries: List[tuple]) -> List[Any]:
           """Execute multiple queries in a transaction with optimization"""
           start_time = time.time()
           results = []
           
           try:
               async with self.get_connection() as conn:
                   async with conn.transaction():
                       for query, args in queries:
                           if isinstance(args, dict) and args.get('fetch_all'):
                               result = await conn.fetch(query, *args.get('params', []))
                           elif isinstance(args, dict) and args.get('fetch_one'):
                               result = await conn.fetchrow(query, *args.get('params', []))
                           else:
                               if isinstance(args, dict):
                                   args = args.get('params', [])
                               result = await conn.execute(query, *args)
                           results.append(result)
                   
                   execution_time = time.time() - start_time
                   logger.debug(f"Transaction completed in {execution_time:.3f}s ({len(queries)} queries)")
                   
                   return results
                   
           except Exception as e:
               execution_time = time.time() - start_time
               logger.error(f"Transaction failed after {execution_time:.3f}s: {e}")
               raise
               
       async def bulk_insert_optimized(
           self, 
           table: str, 
           columns: List[str], 
           data: List[tuple],
           batch_size: int = 1000
       ) -> int:
           """Optimized bulk insert with batching"""
           total_inserted = 0
           
           # Create INSERT query with COPY for better performance
           placeholders = ','.join(['$' + str(i) for i in range(1, len(columns) + 1)])
           insert_query = f"INSERT INTO {table} ({','.join(columns)}) VALUES ({placeholders})"
           
           try:
               async with self.get_connection() as conn:
                   # Use COPY for large datasets (more efficient than INSERT)
                   if len(data) > batch_size:
                       await conn.copy_records_to_table(
                           table, 
                           records=data, 
                           columns=columns
                       )
                       total_inserted = len(data)
                   else:
                       # Use executemany for smaller datasets
                       await conn.executemany(insert_query, data)
                       total_inserted = len(data)
                       
               logger.info(f"Bulk inserted {total_inserted} records into {table}")
               return total_inserted
               
           except Exception as e:
               logger.error(f"Bulk insert failed: {e}")
               raise
               
       async def get_pool_stats(self) -> ConnectionStats:
           """Get detailed pool statistics"""
           if not self.pool:
               return ConnectionStats(0, 0, 0, 0.0, 0, 0)
           
           pool_size = self.pool.get_size()
           active_connections = pool_size - self.pool.get_idle_size()
           idle_connections = self.pool.get_idle_size()
           
           avg_query_time = (
               sum(self.query_metrics) / len(self.query_metrics)
               if self.query_metrics else 0.0
           )
           
           return ConnectionStats(
               total_connections=pool_size,
               active_connections=active_connections,
               idle_connections=idle_connections,
               avg_query_time=avg_query_time,
               total_queries=self.total_queries,
               failed_queries=self.failed_queries
           )
           
       async def _monitor_pool(self):
           """Monitor pool health and performance"""
           while True:
               try:
                   await asyncio.sleep(30)  # Check every 30 seconds
                   
                   stats = await self.get_pool_stats()
                   
                   # Log if pool utilization is high
                   utilization = stats.active_connections / max(1, stats.total_connections)
                   if utilization > 0.8:
                       logger.warning(f"High pool utilization: {utilization:.1%}")
                   
                   # Log if average query time is high
                   if stats.avg_query_time > 0.5:
                       logger.warning(f"High average query time: {stats.avg_query_time:.3f}s")
                   
                   # Log failure rate
                   if stats.total_queries > 0:
                       failure_rate = stats.failed_queries / stats.total_queries
                       if failure_rate > 0.05:  # 5% failure rate
                           logger.warning(f"High query failure rate: {failure_rate:.1%}")
                           
               except asyncio.CancelledError:
                   break
               except Exception as e:
                   logger.error(f"Pool monitoring error: {e}")
                   
       async def health_check(self) -> bool:
           """Check if database pool is healthy"""
           try:
               async with self.get_connection() as conn:
                   await conn.fetchval('SELECT 1')
                   return True
           except Exception as e:
               logger.error(f"Database health check failed: {e}")
               return False
               
       async def close_pool(self):
           """Close database pool gracefully"""
           if self.pool:
               await self.pool.close()
               logger.info("Database pool closed")
   ```

1. **Multi-Layer Caching System**:
   ```python
   # app/core/advanced_cache.py
   import asyncio
   import aioredis
   import json
   import pickle
   import hashlib
   import time
   from typing import Any, Dict, List, Optional, Union, Callable
   from dataclasses import dataclass
   from enum import Enum
   import logging
   
   logger = logging.getLogger(__name__)
   
   class CacheLevel(Enum):
       MEMORY = "memory"
       REDIS = "redis"
       DATABASE = "database"
   
   @dataclass
   class CacheEntry:
       key: str
       value: Any
       ttl: int
       created_at: float
       access_count: int = 0
       last_accessed: float = 0
       
   class MultiLayerCache:
       def __init__(self, redis_url: str, max_memory_items: int = 1000):
           self.redis_url = redis_url
           self.redis_client = None
           self.memory_cache: Dict[str, CacheEntry] = {}
           self.max_memory_items = max_memory_items
           self.cache_stats = {
               'hits': {'memory': 0, 'redis': 0, 'database': 0},
               'misses': 0,
               'evictions': 0
           }
           
       async def initialize(self):
           """Initialize cache connections"""
           self.redis_client = await aioredis.from_url(
               self.redis_url, 
               decode_responses=False,  # Handle binary data
               max_connections=20
           )
           
           # Start cleanup task for memory cache
           asyncio.create_task(self._cleanup_memory_cache())
           
       async def get(self, key: str, fetch_func: Optional[Callable] = None) -> Any:
           """Get value with multi-layer lookup"""
           cache_key = self._normalize_key(key)
           
           # Level 1: Memory cache
           if cache_key in self.memory_cache:
               entry = self.memory_cache[cache_key]
               if not self._is_expired(entry):
                   entry.access_count += 1
                   entry.last_accessed = time.time()
                   self.cache_stats['hits']['memory'] += 1
                   return entry.value
               else:
                   del self.memory_cache[cache_key]
           
           # Level 2: Redis cache
           try:
               redis_value = await self.redis_client.get(cache_key)
               if redis_value is not None:
                   value = pickle.loads(redis_value)
                   
                   # Promote to memory cache
                   await self._set_memory_cache(cache_key, value, 300)  # 5 min in memory
                   
                   self.cache_stats['hits']['redis'] += 1
                   return value
           except Exception as e:
               logger.error(f"Redis cache error: {e}")
           
           # Level 3: Fetch from source (database/API)
           if fetch_func:
               try:
                   value = await fetch_func() if asyncio.iscoroutinefunction(fetch_func) else fetch_func()
                   
                   # Cache at multiple levels
                   await self.set(cache_key, value, ttl=600)  # 10 min default
                   
                   self.cache_stats['hits']['database'] += 1
                   return value
               except Exception as e:
                   logger.error(f"Fetch function error: {e}")
                   raise
           
           # Cache miss
           self.cache_stats['misses'] += 1
           return None
           
       async def set(
           self, 
           key: str, 
           value: Any, 
           ttl: int = 600,
           memory_ttl: Optional[int] = None
       ):
           """Set value in multi-layer cache"""
           cache_key = self._normalize_key(key)
           
           # Set in Redis
           try:
               serialized_value = pickle.dumps(value)
               await self.redis_client.setex(cache_key, ttl, serialized_value)
           except Exception as e:
               logger.error(f"Redis set error: {e}")
           
           # Set in memory cache (shorter TTL for memory efficiency)
           memory_ttl = memory_ttl or min(ttl, 300)  # Max 5 minutes in memory
           await self._set_memory_cache(cache_key, value, memory_ttl)
           
       async def _set_memory_cache(self, key: str, value: Any, ttl: int):
           """Set value in memory cache with LRU eviction"""
           current_time = time.time()
           
           # Evict if at capacity
           if len(self.memory_cache) >= self.max_memory_items:
               await self._evict_lru()
           
           self.memory_cache[key] = CacheEntry(
               key=key,
               value=value,
               ttl=ttl,
               created_at=current_time,
               last_accessed=current_time
           )
           
       async def _evict_lru(self):
           """Evict least recently used items from memory cache"""
           if not self.memory_cache:
               return
           
           # Find LRU item
           lru_key = min(
               self.memory_cache.keys(),
               key=lambda k: self.memory_cache[k].last_accessed
           )
           
           del self.memory_cache[lru_key]
           self.cache_stats['evictions'] += 1
           
       async def delete(self, key: str):
           """Delete from all cache levels"""
           cache_key = self._normalize_key(key)
           
           # Delete from memory
           if cache_key in self.memory_cache:
               del self.memory_cache[cache_key]
           
           # Delete from Redis
           try:
               await self.redis_client.delete(cache_key)
           except Exception as e:
               logger.error(f"Redis delete error: {e}")
               
       async def delete_pattern(self, pattern: str):
           """Delete keys matching pattern from all cache levels"""
           # Delete from memory cache
           keys_to_delete = [
               key for key in self.memory_cache.keys()
               if pattern in key
           ]
           for key in keys_to_delete:
               del self.memory_cache[key]
           
           # Delete from Redis
           try:
               keys = await self.redis_client.keys(f"*{pattern}*")
               if keys:
                   await self.redis_client.delete(*keys)
           except Exception as e:
               logger.error(f"Redis pattern delete error: {e}")
               
       async def get_or_set(
           self, 
           key: str, 
           fetch_func: Callable,
           ttl: int = 600
       ) -> Any:
           """Get value or set if not exists (cache-aside pattern)"""
           value = await self.get(key)
           
           if value is None:
               value = await fetch_func() if asyncio.iscoroutinefunction(fetch_func) else fetch_func()
               if value is not None:
                   await self.set(key, value, ttl)
           
           return value
           
       async def bulk_get(self, keys: List[str]) -> Dict[str, Any]:
           """Get multiple values efficiently"""
           results = {}
           missing_keys = []
           
           # Check memory cache first
           for key in keys:
               cache_key = self._normalize_key(key)
               if cache_key in self.memory_cache:
                   entry = self.memory_cache[cache_key]
                   if not self._is_expired(entry):
                       results[key] = entry.value
                       entry.access_count += 1
                       entry.last_accessed = time.time()
                   else:
                       del self.memory_cache[cache_key]
                       missing_keys.append(key)
               else:
                   missing_keys.append(key)
           
           # Check Redis for missing keys
           if missing_keys:
               try:
                   redis_keys = [self._normalize_key(key) for key in missing_keys]
                   redis_values = await self.redis_client.mget(redis_keys)
                   
                   for key, redis_value in zip(missing_keys, redis_values):
                       if redis_value is not None:
                           value = pickle.loads(redis_value)
                           results[key] = value
                           
                           # Promote to memory cache
                           await self._set_memory_cache(
                               self._normalize_key(key), value, 300
                           )
               except Exception as e:
                   logger.error(f"Redis bulk get error: {e}")
           
           return results
           
       async def bulk_set(self, items: Dict[str, Any], ttl: int = 600):
           """Set multiple values efficiently"""
           # Set in Redis using pipeline
           try:
               pipe = self.redis_client.pipeline()
               for key, value in items.items():
                   cache_key = self._normalize_key(key)
                   serialized_value = pickle.dumps(value)
                   pipe.setex(cache_key, ttl, serialized_value)
               await pipe.execute()
           except Exception as e:
               logger.error(f"Redis bulk set error: {e}")
           
           # Set in memory cache
           memory_ttl = min(ttl, 300)
           for key, value in items.items():
               await self._set_memory_cache(
                   self._normalize_key(key), value, memory_ttl
               )
               
       def _normalize_key(self, key: str) -> str:
           """Normalize cache key"""
           return f"zergo_qr:{hashlib.sha256(key.encode()).hexdigest()[:16]}"
           
       def _is_expired(self, entry: CacheEntry) -> bool:
           """Check if cache entry is expired"""
           return time.time() > (entry.created_at + entry.ttl)
           
       async def _cleanup_memory_cache(self):
           """Periodic cleanup of expired memory cache entries"""
           while True:
               try:
                   await asyncio.sleep(60)  # Cleanup every minute
                   
                   current_time = time.time()
                   expired_keys = [
                       key for key, entry in self.memory_cache.items()
                       if current_time > (entry.created_at + entry.ttl)
                   ]
                   
                   for key in expired_keys:
                       del self.memory_cache[key]
                   
                   if expired_keys:
                       logger.debug(f"Cleaned up {len(expired_keys)} expired cache entries")
                       
               except asyncio.CancelledError:
                   break
               except Exception as e:
                   logger.error(f"Cache cleanup error: {e}")
                   
       async def get_stats(self) -> Dict[str, Any]:
           """Get comprehensive cache statistics"""
           total_hits = sum(self.cache_stats['hits'].values())
           total_requests = total_hits + self.cache_stats['misses']
           
           hit_rate = total_hits / max(1, total_requests)
           
           return {
               'hit_rate': hit_rate,
               'total_requests': total_requests,
               'hits_by_level': self.cache_stats['hits'],
               'misses': self.cache_stats['misses'],
               'memory_cache_size': len(self.memory_cache),
               'evictions': self.cache_stats['evictions'],
               'redis_info': await self._get_redis_info()
           }
           
       async def _get_redis_info(self) -> Dict[str, Any]:
           """Get Redis connection information"""
           try:
               info = await self.redis_client.info()
               return {
                   'connected_clients': info.get('connected_clients'),
                   'used_memory_human': info.get('used_memory_human'),
                   'keyspace_hits': info.get('keyspace_hits'),
                   'keyspace_misses': info.get('keyspace_misses')
               }
           except Exception:
               return {}
               
       async def close(self):
           """Close cache connections"""
           if self.redis_client:
               await self.redis_client.close()
   ```
