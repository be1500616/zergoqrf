   import 'package:mockito/mockito.dart';
   import 'package:mockito/annotations.dart';
   import 'package:get/get.dart';

   @GenerateMocks([SupabaseProvider])
   import 'auth_service_test.mocks.dart';
   
   void main() {
     group('AuthService Tests', () {
       late AuthService authService;
       late MockSupabaseProvider mockSupabaseProvider;
       
       setUp(() {
         Get.testMode = true;
         mockSupabaseProvider = MockSupabaseProvider();
         authService = AuthService();
         Get.put<SupabaseProvider>(mockSupabaseProvider);
       });
       
       tearDown(() {
         Get.reset();
       });
       
       group('Login Tests', () {
         test('should return success when login credentials are valid', () async {
           // Arrange
           const email = 'test@example.com';
           const password = 'password123';
           final mockUser = UserModel(
             id: '123',
             email: email,
             fullName: 'Test User',
             phoneNumber: '+1234567890',
           );
           
           when(mockSupabaseProvider.signInWithEmailAndPassword(
             email: email,
             password: password,
           )).thenAnswer((_) async => mockUser);
           
           // Act
           final result = await authService.login(email, password);
           
           // Assert
           expect(result.isSuccess, true);
           expect(result.data, equals(mockUser));
           verify(mockSupabaseProvider.signInWithEmailAndPassword(
             email: email,
             password: password,
           )).called(1);
         });
         
         test('should return error when login credentials are invalid', () async {
           // Arrange
           const email = 'invalid@example.com';
           const password = 'wrongpassword';
           
           when(mockSupabaseProvider.signInWithEmailAndPassword(
             email: email,
             password: password,
           )).thenThrow(Exception('Invalid credentials'));
           
           // Act
           final result = await authService.login(email, password);
           
           // Assert
           expect(result.isSuccess, false);
           expect(result.error, contains('Invalid credentials'));
         });
         
         test('should validate email format before login attempt', () async {
           // Arrange
           const invalidEmail = 'invalid-email';
           const password = 'password123';
           
           // Act
           final result = await authService.login(invalidEmail, password);
           
           // Assert
           expect(result.isSuccess, false);
           expect(result.error, contains('Invalid email format'));
           verifyNever(mockSupabaseProvider.signInWithEmailAndPassword(
             email: anyNamed('email'),
             password: anyNamed('password'),
           ));
         });
       });
       
       group('Registration Tests', () {
         test('should register user successfully with valid data', () async {
           // Arrange
           final registrationData = {
             'email': 'newuser@example.com',
             'password': 'SecurePass123!',
             'fullName': 'New User',
             'phoneNumber': '+1234567890',
           };
           
           final mockUser = UserModel.fromJson({
             'id': '456',
             ...registrationData,
           });
           
           when(mockSupabaseProvider.signUp(any))
               .thenAnswer((_) async => mockUser);
           
           // Act
           final result = await authService.register(registrationData);
           
           // Assert
           expect(result.isSuccess, true);
           expect(result.data.email, equals(registrationData['email']));
         });
         
         test('should validate password strength during registration', () async {
           // Arrange
           final weakPasswordData = {
             'email': 'user@example.com',
             'password': '123', // Weak password
             'fullName': 'Test User',
           };
           
           // Act
           final result = await authService.register(weakPasswordData);
           
           // Assert
           expect(result.isSuccess, false);
           expect(result.error, contains('Password must be at least 8 characters'));
         });
       });
       
       group('Token Management Tests', () {
         test('should refresh token when expired', () async {
           // Arrange
           const expiredToken = 'expired.jwt.token';
           const newToken = 'new.jwt.token';
           
           when(mockSupabaseProvider.refreshToken())
               .thenAnswer((_) async => newToken);
           
           // Act
           final result = await authService.refreshAuthToken();
           
           // Assert
           expect(result, equals(newToken));
           verify(mockSupabaseProvider.refreshToken()).called(1);
         });
         
         test('should handle token refresh failure gracefully', () async {
           // Arrange
           when(mockSupabaseProvider.refreshToken())
               .thenThrow(Exception('Token refresh failed'));
           
           // Act & Assert
           expect(
             () async => await authService.refreshAuthToken(),
             throwsA(isA<Exception>()),
           );
         });
       });
     });
   }
