1. **Data Warehouse Design and ETL Pipeline**:
   ```python
   # apps/backend/app/features/analytics/infrastructure/data_warehouse.py
   from sqlalchemy import create_engine, Column, Integer, String, DateTime, Decimal, ForeignKey
   from sqlalchemy.ext.declarative import declarative_base
   from sqlalchemy.orm import relationship
   import pandas as pd
   from datetime import datetime, timedelta
   
   Base = declarative_base()
   
   class FactSales(Base):
       """Fact table for sales transactions"""
       __tablename__ = 'fact_sales'
       
       id = Column(Integer, primary_key=True)
       order_id = Column(String, unique=True)
       restaurant_id = Column(String, ForeignKey('dim_restaurant.id'))
       customer_id = Column(String, ForeignKey('dim_customer.id'))
       menu_item_id = Column(String, ForeignKey('dim_menu_item.id'))
       date_key = Column(Integer, ForeignKey('dim_date.date_key'))
       time_key = Column(Integer, ForeignKey('dim_time.time_key'))
       
       # Measures
       quantity = Column(Integer)
       unit_price = Column(Decimal(10, 2))
       total_amount = Column(Decimal(10, 2))
       discount_amount = Column(Decimal(10, 2))
       tax_amount = Column(Decimal(10, 2))
       
       # Foreign key relationships
       restaurant = relationship("DimRestaurant")
       customer = relationship("DimCustomer")
       menu_item = relationship("DimMenuItem")
       date = relationship("DimDate")
       time = relationship("DimTime")
   
   class AnalyticsETLService:
       def __init__(self, source_db_url: str, warehouse_db_url: str):
           self.source_engine = create_engine(source_db_url)
           self.warehouse_engine = create_engine(warehouse_db_url)
           
       async def extract_transform_load_sales_data(self, date_range: tuple):
           """ETL process for sales data"""
           start_date, end_date = date_range
           
           # Extract data from operational database
           sales_query = """
           SELECT 
               o.id as order_id,
               o.restaurant_id,
               o.customer_id,
               oi.menu_item_id,
               oi.quantity,
               oi.unit_price,
               oi.total_amount,
               o.discount_amount,
               o.tax_amount,
               o.created_at
           FROM orders o
           JOIN order_items oi ON o.id = oi.order_id
           WHERE o.created_at BETWEEN %s AND %s
           AND o.status = 'completed'
           """
           
           sales_df = pd.read_sql(sales_query, self.source_engine, params=(start_date, end_date))
           
           # Transform data
           sales_df['date_key'] = sales_df['created_at'].dt.strftime('%Y%m%d').astype(int)
           sales_df['time_key'] = (sales_df['created_at'].dt.hour * 100 + 
                                  sales_df['created_at'].dt.minute)
           
           # Load to data warehouse
           sales_df.to_sql('fact_sales', self.warehouse_engine, if_exists='append', index=False)
   ```

2. **Real-time Analytics Processing**:
   ```python
   # apps/backend/app/features/analytics/application/real_time_analytics.py
   import asyncio
   import json
   from datetime import datetime, timedelta
   from typing import Dict, List
   
   class RealTimeAnalyticsService:
       def __init__(self, redis_client, websocket_manager):
           self.redis = redis_client
           self.websocket_manager = websocket_manager
           self.analytics_cache = {}
           
       async def process_order_event(self, order_data: dict):
           """Process real-time order events for analytics"""
           restaurant_id = order_data['restaurant_id']
           
           # Update real-time metrics
           await self.update_realtime_metrics(restaurant_id, order_data)
           
           # Send updates to connected dashboards
           await self.broadcast_metrics_update(restaurant_id)
           
       async def update_realtime_metrics(self, restaurant_id: str, order_data: dict):
           """Update real-time metrics in Redis"""
           current_hour = datetime.now().strftime('%Y%m%d%H')
           
           # Hourly sales metrics
           hourly_key = f"analytics:hourly:{restaurant_id}:{current_hour}"
           await self.redis.hincrby(hourly_key, "orders_count", 1)
           await self.redis.hincrbyfloat(hourly_key, "revenue", float(order_data['total_amount']))
           await self.redis.expire(hourly_key, 86400)  # Expire after 24 hours
           
       async def get_live_dashboard_data(self, restaurant_id: str) -> Dict:
           """Get live dashboard data for restaurant"""
           current_hour = datetime.now().strftime('%Y%m%d%H')
           current_date = current_hour[:8]
           
           # Get hourly data
           hourly_key = f"analytics:hourly:{restaurant_id}:{current_hour}"
           hourly_data = await self.redis.hgetall(hourly_key)
           
           # Get daily data
           daily_key = f"analytics:daily:{restaurant_id}:{current_date}"
           daily_data = await self.redis.hgetall(daily_key)
           
           return {
               'hourly_metrics': {
                   'orders': int(hourly_data.get('orders_count', 0)),
                   'revenue': float(hourly_data.get('revenue', 0)),
               },
               'daily_metrics': {
                   'orders': int(daily_data.get('orders_count', 0)),
                   'revenue': float(daily_data.get('revenue', 0)),
               },
               'timestamp': datetime.now().isoformat()
           }
   ```

   # apps/backend/app/features/analytics/application/predictive_analytics.py
   import pandas as pd
   import numpy as np
   from sklearn.ensemble import RandomForestRegressor
   from sklearn.preprocessing import StandardScaler
   from datetime import datetime, timedelta
   
   class PredictiveAnalyticsService:
       def __init__(self):
           self.models = {}
           self.scalers = {}
           
       async def predict_demand_forecasting(self, restaurant_id: str, days_ahead: int = 7):
           """Predict demand for menu items for upcoming days"""
           
           # Get historical data
           historical_data = await self._get_historical_sales_data(restaurant_id, days=90)
           
           if len(historical_data) < 30:  # Need at least 30 days of data
               return {"error": "Insufficient historical data for prediction"}
           
           # Prepare features
           features_df = self._prepare_demand_features(historical_data)
           
           # Train or load model
           model_key = f"demand_forecast_{restaurant_id}"
           if model_key not in self.models:
               await self._train_demand_model(model_key, features_df)
           
           # Generate predictions
           predictions = []
           base_date = datetime.now().date()
           
           for day in range(1, days_ahead + 1):
               target_date = base_date + timedelta(days=day)
               prediction_features = self._create_prediction_features(target_date, historical_data)
               
               predicted_demand = self.models[model_key].predict([prediction_features])[0]
               
               predictions.append({
                   'date': target_date.isoformat(),
                   'predicted_orders': max(0, int(predicted_demand)),
                   'day_of_week': target_date.strftime('%A'),
                   'confidence_interval': self._calculate_confidence_interval(predicted_demand)
               })
           
           return {
               'restaurant_id': restaurant_id,
               'forecast_period': f"{days_ahead} days",
               'predictions': predictions,
               'model_accuracy': self._get_model_accuracy(model_key),
               'generated_at': datetime.utcnow().isoformat()
           }
