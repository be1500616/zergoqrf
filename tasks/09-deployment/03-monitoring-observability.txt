   # apps/backend/app/core/monitoring.py
   import logging
   import time
   from functools import wraps
   from typing import Dict, Any
   from datadog import initialize, statsd
   from ddtrace import tracer, patch_all
   import psutil
   import asyncio
   
   # Initialize DataDog
   initialize(
       api_key=settings.DATADOG_API_KEY,
       app_key=settings.DATADOG_APP_KEY
   )
   
   # Auto-instrument common libraries
   patch_all()
   
   class MetricsCollector:
       def __init__(self):
           self.namespace = "zergo_qr"
           
       def increment_counter(self, metric_name: str, value: int = 1, tags: list = None):
           """Increment a counter metric"""
           statsd.increment(
               f"{self.namespace}.{metric_name}",
               value=value,
               tags=tags or []
           )
           
       def record_gauge(self, metric_name: str, value: float, tags: list = None):
           """Record a gauge metric"""
           statsd.gauge(
               f"{self.namespace}.{metric_name}",
               value=value,
               tags=tags or []
           )
           
       def record_histogram(self, metric_name: str, value: float, tags: list = None):
           """Record a histogram metric"""
           statsd.histogram(
               f"{self.namespace}.{metric_name}",
               value=value,
               tags=tags or []
           )
           
       def time_function(self, metric_name: str, tags: list = None):
           """Decorator to time function execution"""
           def decorator(func):
               @wraps(func)
               async def async_wrapper(*args, **kwargs):
                   start_time = time.time()
                   try:
                       result = await func(*args, **kwargs)
                       self.record_histogram(
                           f"{metric_name}.duration",
                           time.time() - start_time,
                           tags=(tags or []) + ["status:success"]
                       )
                       return result
                   except Exception as e:
                       self.record_histogram(
                           f"{metric_name}.duration",
                           time.time() - start_time,
                           tags=(tags or []) + ["status:error"]
                       )
                       raise
                       
               @wraps(func)
               def sync_wrapper(*args, **kwargs):
                   start_time = time.time()
                   try:
                       result = func(*args, **kwargs)
                       self.record_histogram(
                           f"{metric_name}.duration",
                           time.time() - start_time,
                           tags=(tags or []) + ["status:success"]
                       )
                       return result
                   except Exception as e:
                       self.record_histogram(
                           f"{metric_name}.duration",
                           time.time() - start_time,
                           tags=(tags or []) + ["status:error"]
                       )
                       raise
                       
               return async_wrapper if asyncio.iscoroutinefunction(func) else sync_wrapper
           return decorator
   
   metrics = MetricsCollector()
   
   class SystemMetricsCollector:
       def __init__(self):
           self.metrics = MetricsCollector()
           
       async def collect_system_metrics(self):
           """Collect and send system metrics"""
           while True:
               try:
                   # CPU metrics
                   cpu_percent = psutil.cpu_percent(interval=1)
                   self.metrics.record_gauge("system.cpu.usage", cpu_percent)
                   
                   # Memory metrics
                   memory = psutil.virtual_memory()
                   self.metrics.record_gauge("system.memory.usage", memory.percent)
                   self.metrics.record_gauge("system.memory.available", memory.available)
                   
                   # Disk metrics
                   disk = psutil.disk_usage('/')
                   self.metrics.record_gauge("system.disk.usage", disk.percent)
                   self.metrics.record_gauge("system.disk.free", disk.free)
                   
                   # Network metrics
                   network = psutil.net_io_counters()
                   self.metrics.record_gauge("system.network.bytes_sent", network.bytes_sent)
                   self.metrics.record_gauge("system.network.bytes_recv", network.bytes_recv)
                   
                   await asyncio.sleep(30)  # Collect every 30 seconds
                   
               except Exception as e:
                   logger.error(f"Error collecting system metrics: {e}")
                   await asyncio.sleep(60)

2. **Custom Middleware for Request Monitoring**:
   ```python
   # apps/backend/app/core/monitoring_middleware.py
   from fastapi import Request, Response
   from starlette.middleware.base import BaseHTTPMiddleware
   import time
   import uuid
   
   class MonitoringMiddleware(BaseHTTPMiddleware):
       def __init__(self, app):
           super().__init__(app)
           self.metrics = MetricsCollector()
           
       async def dispatch(self, request: Request, call_next):
           # Generate request ID
           request_id = str(uuid.uuid4())
           request.state.request_id = request_id
           
           # Start timing
           start_time = time.time()
           
           # Extract request info
           method = request.method
           path = request.url.path
           user_agent = request.headers.get("user-agent", "unknown")
           client_ip = request.client.host
           
           # Create span for tracing
           with tracer.trace(
               "http.request",
               service="zergo-qr-api",
               resource=f"{method} {path}",
               span_type="web"
           ) as span:
               span.set_tag("http.method", method)
               span.set_tag("http.url", str(request.url))
               span.set_tag("user.ip", client_ip)
               span.set_tag("request.id", request_id)
               
               try:
                   # Process request
                   response = await call_next(request)
                   
                   # Calculate duration
                   duration = time.time() - start_time
                   
                   # Set span tags
                   span.set_tag("http.status_code", response.status_code)
                   span.set_tag("response.duration", duration)
                   
                   # Record metrics
                   tags = [
                       f"method:{method}",
                       f"endpoint:{path}",
                       f"status_code:{response.status_code}"
                   ]
                   
                   # Request count
                   self.metrics.increment_counter("api.requests.total", tags=tags)
                   
                   # Response time
                   self.metrics.record_histogram("api.response_time", duration, tags=tags)
                   
                   # Error rate
                   if response.status_code >= 400:
                       self.metrics.increment_counter("api.errors.total", tags=tags)
                       span.set_tag("error", True)
                   
                   # Add response headers
                   response.headers["X-Request-ID"] = request_id
                   response.headers["X-Response-Time"] = f"{duration:.3f}s"
                   
                   return response
                   
               except Exception as e:
                   duration = time.time() - start_time
                   
                   # Record error metrics
                   error_tags = [
                       f"method:{method}",
                       f"endpoint:{path}",
                       f"error_type:{type(e).__name__}"
                   ]
                   
                   self.metrics.increment_counter("api.errors.total", tags=error_tags)
                   self.metrics.record_histogram("api.response_time", duration, tags=error_tags)
                   
                   # Set error span
                   span.set_tag("error", True)
                   span.set_tag("error.message", str(e))
                   span.set_tag("error.type", type(e).__name__)
                   
                   raise
   ```

   class StructuredLogger:
       def __init__(self, name: str):
           self.logger = logging.getLogger(name)
           
       def info(self, message: str, extra: dict = None):
           self.logger.info(message, extra=extra or {})
           
       def error(self, message: str, error: Exception = None, extra: dict = None):
           error_extra = extra or {}
           if error:
               error_extra.update({
                   "error_type": type(error).__name__,
                   "error_message": str(error)
               })
           self.logger.error(message, extra=error_extra)
           
       def warning(self, message: str, extra: dict = None):
           self.logger.warning(message, extra=extra or {})

1. **Comprehensive Health Check System**:
   ```python
   # apps/backend/app/core/health_checks.py
   from fastapi import APIRouter, HTTPException
   from typing import Dict, Any
   import asyncio
   import time
   from enum import Enum
   
   class HealthStatus(Enum):
       HEALTHY = "healthy"
       DEGRADED = "degraded"
       UNHEALTHY = "unhealthy"
   
   class HealthChecker:
       def __init__(self):
           self.checks = {}
           self.metrics = MetricsCollector()
           
       def register_check(self, name: str, check_func):
           """Register a health check function"""
           self.checks[name] = check_func
           
       async def run_all_checks(self) -> Dict[str, Any]:
           """Run all registered health checks"""
           results = {}
           overall_status = HealthStatus.HEALTHY
           start_time = time.time()
           
           for name, check_func in self.checks.items():
               try:
                   check_start = time.time()
                   result = await check_func()
                   check_duration = time.time() - check_start
                   
                   results[name] = {
                       "status": result.get("status", HealthStatus.HEALTHY.value),
                       "details": result.get("details", {}),
                       "duration_ms": round(check_duration * 1000, 2)
                   }
                   
                   # Record metrics
                   self.metrics.record_histogram(
                       f"health_check.{name}.duration",
                       check_duration,
                       tags=[f"status:{result.get('status', 'healthy')}"]
                   )
                   
                   # Determine overall status
                   check_status = HealthStatus(result.get("status", HealthStatus.HEALTHY.value))
                   if check_status == HealthStatus.UNHEALTHY:
                       overall_status = HealthStatus.UNHEALTHY
                   elif check_status == HealthStatus.DEGRADED and overall_status == HealthStatus.HEALTHY:
                       overall_status = HealthStatus.DEGRADED
                       
               except Exception as e:
                   check_duration = time.time() - check_start
                   results[name] = {
                       "status": HealthStatus.UNHEALTHY.value,
                       "details": {"error": str(e)},
                       "duration_ms": round(check_duration * 1000, 2)
                   }
                   overall_status = HealthStatus.UNHEALTHY
                   
                   # Record error metrics
                   self.metrics.increment_counter(
                       f"health_check.{name}.errors",
                       tags=["error_type:exception"]
                   )
           
           total_duration = time.time() - start_time
           
           return {
               "status": overall_status.value,
               "timestamp": datetime.utcnow().isoformat(),
               "duration_ms": round(total_duration * 1000, 2),
               "checks": results
           }
   
   # Initialize health checker
   health_checker = HealthChecker()
   
   # Database health check
   async def check_database():
       try:
           async with get_database_session() as session:
               await session.execute(text("SELECT 1"))
               return {"status": HealthStatus.HEALTHY.value}
       except Exception as e:
           return {
               "status": HealthStatus.UNHEALTHY.value,
               "details": {"error": str(e)}
           }
   
   # Redis health check
   async def check_redis():
       try:
           await redis_client.ping()
           return {"status": HealthStatus.HEALTHY.value}
       except Exception as e:
           return {
               "status": HealthStatus.UNHEALTHY.value,
               "details": {"error": str(e)}
           }
   
   # Supabase health check
   async def check_supabase():
       try:
           response = await supabase_client.client.table('health_check').select('*').limit(1).execute()
           return {"status": HealthStatus.HEALTHY.value}
       except Exception as e:
           return {
               "status": HealthStatus.DEGRADED.value,
               "details": {"error": str(e)}
           }
   
   # Register health checks
   health_checker.register_check("database", check_database)
   health_checker.register_check("redis", check_redis)
   health_checker.register_check("supabase", check_supabase)
   
   # Health check router
   router = APIRouter()
   
   @router.get("/health")
   async def health_check():
       """Basic health check endpoint"""
       return {
           "status": "healthy",
           "timestamp": datetime.utcnow().isoformat(),
           "version": "1.0.0"
       }
   
   @router.get("/health/detailed")
   async def detailed_health_check():
       """Detailed health check with all components"""
       result = await health_checker.run_all_checks()
       
       if result["status"] == HealthStatus.UNHEALTHY.value:
           raise HTTPException(status_code=503, detail=result)
       elif result["status"] == HealthStatus.DEGRADED.value:
           raise HTTPException(status_code=200, detail=result)
       
       return result
   ```
