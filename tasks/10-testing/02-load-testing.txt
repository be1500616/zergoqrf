       async def test_database_queries(self, concurrent_connections: int, duration_seconds: int):
           """Test database with multiple concurrent connections"""
           
           async def worker(worker_id: int):
               conn = await asyncpg.connect(**self.db_config)
               start_time = time.time()
               
               while time.time() - start_time < duration_seconds:
                   query_start = time.time()
                   
                   try:
                       # Simulate various query patterns

                           # Menu browsing query

                               FROM menu_items m
                               JOIN restaurants r ON m.restaurant_id = r.id
                               WHERE r.id = $1
                               ORDER BY m.category, m.name
                               LIMIT 20
                           """, f"test_rest_{worker_id % 100}")
                       
                       elif query_type == 1:
                           # Order creation simulation
                           result = await conn.fetchrow("""
                               INSERT INTO orders (id, user_id, restaurant_id, status, total_amount, created_at)
                               VALUES ($1, $2, $3, 'pending', $4, NOW())

                           """, f"order_{worker_id}_{int(time.time())}",

                       elif query_type == 2:
                           # Complex analytics query

                                   COUNT(o.id) as order_count,
                                   SUM(o.total_amount) as total_revenue,
                                   AVG(o.total_amount) as avg_order_value
                               FROM restaurants r
                               LEFT JOIN orders o ON r.id = o.restaurant_id
                               WHERE o.created_at >= NOW() - INTERVAL '24 hours'
                               GROUP BY r.id, r.name
                               ORDER BY total_revenue DESC
                               LIMIT 10

                               FROM users u
                               LEFT JOIN orders o ON u.id = o.user_id
                               WHERE u.id = $1
                               GROUP BY u.id
                           """, f"user_{worker_id % 1000}")
                       
                       query_time = time.time() - query_start

                           'query_type': query_type,
                           'duration': query_time,
                           'timestamp': time.time(),

                       self.results['errors'].append({

                           'error': str(e),

                           await redis.set(f"test_key_{worker_id}_{int(time.time())}", 
                                         f"test_value_{worker_id}", ex=3600)
                       
                       elif operation_type == 1:
                           # Get operation
                           await redis.get(f"test_key_{worker_id}_{int(time.time() - 10)}")
                       
                       elif operation_type == 2:
                           # List operations
                           list_key = f"test_list_{worker_id}"
                           await redis.lpush(list_key, f"item_{int(time.time())}")
                           await redis.ltrim(list_key, 0, 100)  # Keep only latest 100 items
                       
                       elif operation_type == 3:
                           # Hash operations
                           hash_key = f"test_hash_{worker_id}"
                           await redis.hset(hash_key, f"field_{int(time.time())}", 
                                          f"value_{worker_id}")
                       
                       else:
                           # Set operations (for rate limiting simulation)
                           window_key = f"rate_limit:{worker_id}:{int(time.time() / 60)}"
                           await redis.incr(window_key)
                           await redis.expire(window_key, 60)
                       
                       operation_time = time.time() - operation_start
                       self.results['queries'].append({
                           'worker_id': worker_id,
                           'operation_type': f"redis_{operation_type}",
                           'duration': operation_time,
                           'timestamp': time.time(),
                           'success': True,
                       })
                       
                   except Exception as e:
                       operation_time = time.time() - operation_start
                       self.results['errors'].append({
                           'worker_id': worker_id,
                           'error': str(e),
                           'duration': operation_time,
                           'timestamp': time.time(),
                       })
                   
                   await asyncio.sleep(0.05)
               
               await redis.close()
           
           tasks = [redis_worker(i) for i in range(concurrent_connections)]
           await asyncio.gather(*tasks)
   
       async def monitor_system_resources(self, duration_seconds: int):
           """Monitor system CPU and memory usage"""
           start_time = time.time()
           
           while time.time() - start_time < duration_seconds:
               try:
                   cpu_percent = psutil.cpu_percent(interval=1)
                   memory = psutil.virtual_memory()
                   
                   self.results['cpu_usage'].append({
                       'timestamp': time.time(),
                       'cpu_percent': cpu_percent,
                   })
                   
                   self.results['memory_usage'].append({
                       'timestamp': time.time(),
                       'memory_percent': memory.percent,
                       'memory_used_gb': memory.used / (1024**3),
                       'memory_available_gb': memory.available / (1024**3),
                   })
                   
               except Exception as e:
                   logging.error(f"Error monitoring system resources: {e}")
               
               await asyncio.sleep(5)
   
       def generate_report(self) -> Dict[str, Any]:
           """Generate comprehensive performance report"""
           if not self.results['queries']:
               return {"error": "No query results to analyze"}
           
           query_times = [q['duration'] for q in self.results['queries'] if q['success']]
           
           report = {
               'summary': {
                   'total_queries': len(self.results['queries']),
                   'successful_queries': len([q for q in self.results['queries'] if q['success']]),
                   'failed_queries': len(self.results['errors']),
                   'error_rate': len(self.results['errors']) / len(self.results['queries']) if self.results['queries'] else 0,
               },
               'performance_metrics': {
                   'avg_response_time': statistics.mean(query_times) if query_times else 0,
                   'median_response_time': statistics.median(query_times) if query_times else 0,
                   'p95_response_time': self.percentile(query_times, 95) if query_times else 0,
                   'p99_response_time': self.percentile(query_times, 99) if query_times else 0,
                   'min_response_time': min(query_times) if query_times else 0,
                   'max_response_time': max(query_times) if query_times else 0,
               },
               'system_resources': {
                   'avg_cpu_usage': statistics.mean([r['cpu_percent'] for r in self.results['cpu_usage']]) if self.results['cpu_usage'] else 0,
                   'max_cpu_usage': max([r['cpu_percent'] for r in self.results['cpu_usage']]) if self.results['cpu_usage'] else 0,
                   'avg_memory_usage': statistics.mean([r['memory_percent'] for r in self.results['memory_usage']]) if self.results['memory_usage'] else 0,
                   'max_memory_usage': max([r['memory_percent'] for r in self.results['memory_usage']]) if self.results['memory_usage'] else 0,
               },
               'errors': self.results['errors'][:10],  # Show first 10 errors
           }
           
           return report
   
       @staticmethod
       def percentile(data: List[float], percentile: int) -> float:
           """Calculate percentile value"""
           if not data:
               return 0
           sorted_data = sorted(data)
           index = int(len(sorted_data) * percentile / 100)
           return sorted_data[min(index, len(sorted_data) - 1)]
